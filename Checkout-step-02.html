function findScrollableParent($start) {
const ancestors = $start.parents();
for (let i = 0; i < ancestors.length; i++) {
const el = ancestors[i];
const $el = $(el);
if (!$el.is(':visible')) continue;

// 計算済みスタイルで overflow を判定
const cs = window.getComputedStyle(el);
const overflowY = cs.overflowY;
const overflow = cs.overflow;
const scrollable = (overflowY === 'auto' || overflowY === 'scroll' ||
overflow === 'auto' || overflow === 'scroll');

if (scrollable && el.scrollHeight > el.clientHeight) {
return $el; // ここが本当にスクロールしている箱
}
}

// フォールバック: 現在開いているjQuery Modal配下から候補を探す
const $modalScope = $('.jquery-modal.blocker.current');

const $fallback = $modalScope.find('#modal-content, .modal, .payment-form-inner')
.filter(function () {
const el = this;
if (!$(el).is(':visible')) return false;
const cs = window.getComputedStyle(el);
const oy = cs.overflowY, o = cs.overflow;
const scrollable = (oy === 'auto' || oy === 'scroll' || o === 'auto' || o === 'scroll');
return scrollable && el.scrollHeight > el.clientHeight;
})
.first();
if ($fallback.length) return $fallback;
}
return null;
}

window.getByteLength = function(str) {
let length = 0;
for (let i = 0; i < str.length; i++) {
const charCode = str.charCodeAt(i);
if ((charCode >= 0x00 && charCode <= 0x7f) || (charCode >= 0xff61 && charCode <= 0xff9f)) {
length += 1;
} else {
length += 2;
}
}
return length;
};

if (typeof value === 'number' || (value !== '' && !isNaN(value))) {
value = Number(value).toLocaleString();
}
$(element).text(prefix + value + suffix);
};

window.unicodeToChar = function(str) {
return str.replace(/\\u([\da-fA-F]{4})/g, function(match, grp) {
return String.fromCharCode(parseInt(grp, 16));
});
};

window.charToUnicode = function(str) {
if (!str) return '';
let result = '';
for (let i = 0; i < str.length; i++) {
const c = str.charAt(i);
const code = c.charCodeAt(0);
result += '\\u' + ('000' + code.toString(16)).slice(-4);
}
return result;
};

function validateAllInputs() {
let isValid = true;
$('#basket_products_update input[type="text"], #basket_products_update input[type="number"], #basket_products_update textarea').each(function() {
const $input = $(this);
const regexpTest = $input.data('regexp-test');
const maxLineLength = $input.data('max-line-length');
const itemName = $input.data('item-name');
const maxLines = $input.data('max-lines');
const maxTotalLength = $input.attr('maxlength') ? parseInt($input.attr('maxlength')) : null;
const isRequired = $input.closest('.insert-name-item').length > 0 && $input.closest('.insert-name-item').find('label').hasClass('req-icon');
const value = $input.val();

if (isRequired && (!value || value.trim() === '')) {
showErrorMessage([itemName + 'は必須項目です。']);
isValid = false;
return false;
}

if (regexpTest || maxLineLength !== undefined || maxLines !== undefined || maxTotalLength !== null) {
const regexp = regexpTest ? regexpTest.regexp : null;
const errorMessage = regexpTest ? regexpTest.errorMessage : '';
if (!validateInputWithRegexp($input.val(), regexp, errorMessage, maxLineLength, itemName, maxLines, maxTotalLength)) {
isValid = false;
return false;
}
}
});
return isValid;
}

function validateInputWithRegexp(inputValue, regexp, errorMessage, maxLineLength = null, itemName = '', maxLines = null, maxTotalLength = null) {
if (!regexp && maxLineLength === null && maxLines === null && maxTotalLength === null) {
return true;
}

if (maxLines !== null && inputValue) {
const lines = inputValue.split(/\r?\n/);
if (lines.length > maxLines) {
showErrorMessage([`${itemName}の入力可能行数を超えています。`]);
return false;
}
}

if (maxLineLength !== null && inputValue) {
const lines = inputValue.split(/\r?\n/);
for (let i = 0; i < lines.length; i++) {
const byteLength = window.getByteLength(lines[i]);
if (byteLength > maxLineLength) {
showErrorMessage([`${itemName}の一行の入力可能文字数を超えています。`]);
return false;
}
}
}

if (maxTotalLength !== null && inputValue) {
const newLineCount = (inputValue.match(/\r?\n/g) || []).length;
let totalLength = window.getByteLength(inputValue.replace(/\r?\n/g, ''));
if (totalLength > (maxTotalLength - newLineCount)) {
showErrorMessage([`${itemName}の入力可能文字数を超えています。`]);
return false;
}
}

if (regexp && inputValue) {
const valueForRegexp = inputValue.replace(/(\r\n|\r|\n)/g, '');
if (Array.isArray(regexp)) {
for (let i = 0; i < regexp.length; i++) {
const regex = new RegExp(regexp[i], 'u');
if (regex.test(valueForRegexp)) {
showErrorMessage([errorMessage[i]]);
return false;
}
}
} else {
const regex = new RegExp(regexp, 'u');
if (regex.test(valueForRegexp)) {
showErrorMessage([errorMessage]);
return false;
}
}
}
return true;
}

<script>
$(document).ready(function() {
// デバウンス処理を適用
const debouncedUpdate = debounce(updateFormData, 600);

// テキスト入力のイベントを監視（複数のイベントを統合）
$('.order-notes-block input[type="text"], .order-notes-block textarea').each(function() {
$(this).on('input autocompletecompleted autocomplete', debouncedUpdate)
.on('change', debouncedUpdate);

// オートフィル検知のための追加処理
const observer = new MutationObserver((mutations) => {
mutations.forEach((mutation) => {
if (mutation.type === 'attributes' &&
(mutation.attributeName === 'value' ||
mutation.attributeName === 'data-autocompletecompleted')) {
debouncedUpdate();
}
});
});

observer.observe(this, {
attributes: true,
attributeFilter: ['value', 'data-autocompletecompleted']
});
});

// ラジオボタンの変更を監視
$('.order-notes-block input[type="radio"]').on('change', debouncedUpdate).on('blur', debouncedUpdate);

// チェックボックスの変更を監視
$('.order-notes-block input[type="checkbox"]').on('change', debouncedUpdate).on('blur', debouncedUpdate);

function updateFormData() {
const data = collectFormData();
updateData(data);
}

function collectFormData() {
const data = {};
const orderRemarksIds = [
'order_remark_0','order_remark_1' ];
const orderRemarkFieldMap = {"order_remark_0":"20220811085549","order_remark_1":"20220811091009"};
$.each(orderRemarksIds, function(index, ids) {
const remarkId = orderRemarkFieldMap[ids];
if (remarkId) {
data[remarkId] = Olive.getInputValue(ids);
}
});
let deliveryRemarksIds = [];
$('.order-notes input[name^="delivery_remark_"], .order-notes textarea[name^="delivery_remark_"]').each(function() {
const name = $(this).attr('name');
if (name && !deliveryRemarksIds.includes(name)) {
deliveryRemarksIds.push(name);
}
});
$.each(deliveryRemarksIds, function(index, ids) {
data[ids] = Olive.getInputValue(ids);
});

return data;
}

function updateData(params) {
Olive.post("/checkout/step02/remarksOption?db=gcom043", params, function(data) {
return true;
}, null, null, false, false);
}

// デバウンス関数
function debounce(func, wait) {
let timeout;
return function executedFunction(...args) {
const later = () => {
clearTimeout(timeout);
func(...args);
};
clearTimeout(timeout);
timeout = setTimeout(later, wait);
};
}
});
</script>

$(document).on('click', '#dispose_confirm_exec', function() {
const errorField = $('.payment-notice-caution');
Olive.openLoader(100);

// 商品削除の場合
const productData = $(this).data('productData');
const params = {
"quantity": 0,
"productSystemCode": productData.product_system_code,
"variation1ItemId": productData.variation1_itemid,
"variation2ItemId": productData.variation2_itemid,
"optionCodes": productData.option_codes,
"insertNames": [],
};

// 名入れ情報の収集と数量の計算
const insertNames = [];
let totalQuantity = 0;
const quantityFields = $('input[type="number"][id^="quantity' + productData.basket_no + '-"]');

if (quantityFields.length > 0) {
quantityFields.each(function() {
const quantity = Number($(this).val());
const insertNamesInputs = ($(this).data('insert_names_inputs') || '');

// 名入れ情報が一致する場合はスキップ
if (insertNamesInputs === productData.insert_names_inputs) {
return true; // continue
}

totalQuantity += quantity;
insertNames.push({
"quantity": quantity,
"inputs": insertNamesInputs
});
});
}

// 名入れ情報がある場合は数量を設定
if (insertNames.length > 0) {
params.quantity = totalQuantity;
params.insertNames = insertNames;
}

Olive.post("/checkout/step02/quantity/delete?db=gcom043", params, function(data) {
if (data?.success?.isSuccess) {
if (data.success.messages.length == 0) {
location.reload();
} else {
Olive.closeLoader();
}
return false;
}
if (data?.failure?.isFailure) {
Olive.closeLoader();
}
if (data?.error?.isError) {
location.href = '/checkout/error?db=' + 'gcom043';
return false;
}
}, null, 0, false, false);

location.href = '/checkout/error?db=' + 'gcom043';

}
});
